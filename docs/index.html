function deleteExpiredCarePlans(carePlans) { const now = new Date(2026, 1, 24, 12, 24, 18); // Specified date const thirtyDaysAgo = new Date(now - 30 * 24 * 60 * 60 * 1000); const fortyEightHoursAgo = new Date(now - 48 * 60 * 60 * 1000); return carePlans.filter(plan => { return plan.date < thirtyDaysAgo && plan.lastUpdated < fortyEightHoursAgo; }); } // Usage example: // const activePlans = deleteExpiredCarePlans(allCarePlans);
